heartData$target <- as.factor(heartData$target)

Data Splitting

# Create the split
heartSplit <- heartData %>%
    initial_split(prop = 0.8, strata = target)

# Save the two datasets as objects
heartTrain <- training(heartSplit)
heartTest <- testing(heartSplit)
heartTrain %>% ggplot(aes(x=target, y=age)) +geom_boxplot()

heartTrain %>% ggplot(aes(x = as.factor(cp), fill=target)) + 
geom_bar(position = "dodge")

simpleRec <- heartTrain %>%
    recipe(target ~ cp + thalach + exang + oldpeak + ca) %>%
    step_zv(all_predictors()) %>%
    step_corr(all_predictors())
lgSpec <- logistic_reg()
heartWFlow <- workflow() %>%
    add_recipe(simpleRec) %>%
    add_model(lgSpec)
heartBoot <- bootstraps(heartTrain)
heartWFlowFit <- heartWFlow %>%
    fit_resamples(
        resamples=heartBoot, 
        control = control_resamples(save_pred = TRUE)
    )
collect_metrics(heartWFlowFit)
